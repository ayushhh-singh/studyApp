# The sitemap itself — without this, the blanket "Disallow: /" above (with no
# more-specific Allow matching /sitemap.xml) blocks Googlebot from fetching
# it, which Search Console surfaces as "Couldn't fetch" on the Sitemaps page

Allow: /sitemap.xml$

# ⚑ THE SITE ROOT — and this is why the app's logo was missing beside the site in
# Google results. Google discovers a favicon in exactly two places: /favicon.ico,
# and the `<link rel="icon">` in the HOME PAGE's head. `/favicon.png` was already
# allowed below, but Googlebot had no way to LEARN about it: the only rule
# matching "/" was the blanket `Disallow: /`, so the home page was never fetched
# and its icon declaration never read. Allowing the asset without allowing the
# page that declares it is not enough.
#
# `$` anchors this to the exact root, so it opens up "/" and nothing else — every
# app route still falls to the blanket disallow. It ties `Disallow: /` on length,
# and Google resolves an equal-length tie in favour of the LEAST restrictive rule,
# i.e. Allow. `pnpm --filter web check:seo` asserts that resolution rather than
# trusting this comment.
Allow: /$

Allow: /en/$

Allow: /hi/$
